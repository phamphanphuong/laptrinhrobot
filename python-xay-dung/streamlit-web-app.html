# BUỔI 18: BỐ CỤC CHIA CỘT VÀ NÚT TÍNH TOÁN
import streamlit as st

st.subheader("📐 Nhập Kích Thước & Đơn Giá Thi Công")

# Chia thành 2 cột cân đối
col_trai, col_phai = st.columns(2)

with col_trai:
    chieu_dai = st.number_input("Chiều dài (feet):", value=22.0, min_value=0.0)
    chieu_rong = st.number_input("Chiều rộng (feet):", value=16.0, min_value=0.0)

with col_phai:
    don_gia = st.number_input("Đơn giá vật tư (USD/sqft):", value=5.0, min_value=0.0)
    phi_van_chuyen = st.number_input("Phí giao hàng (USD):", value=80.0, min_value=0.0)

# Nút bấm hành động
if st.button("🚀 TÍNH TOÁN BÁO GIÁ NGAY", use_container_width=True):
    dien_tich = chieu_dai * chieu_rong
    if dien_tich > 0:
        tong_tien = (dien_tich * don_gia) + phi_van_chuyen
        st.success(f"🎉 Tổng dự toán công trình: **${tong_tien:,.2f} USD**")
        st.caption(f"Chi tiết: {dien_tich:,.1f} sqft × ${don_gia:.2f} + ${phi_van_chuyen:.2f} vận chuyển.")
    else:
        st.warning("⚠️ Chiều dài hoặc chiều rộng phải lớn hơn 0!")
# Trải nghiệm giao diện: [Cột 1: Dài / Rộng] | [Cột 2: Đơn giá / Vận chuyển] [ Nút bấm full-width: TÍNH TOÁN BÁO GIÁ NGAY ] Box xanh hiển thị: $1,840.00 USD

# BUỔI 19: WEB BÁO GIÁ LÁT SÀN HOÀN CHỈNH
import streamlit as st

st.title("🏡 Dự Toán Lát Sàn Nhà Cải Tạo")

col1, col2 = st.columns(2)
with col1:
    sqft = st.number_input("Tổng diện tích lát sàn (sqft):", value=1200.0, step=50.0)
with col2:
    don_gia_goc = st.number_input("Đơn giá niêm yết (USD/sqft):", value=5.0, step=0.25)

is_nha_cu = st.checkbox("🏚️ Công trình là nhà cũ (cộng 10% phí dọn xà bần)")

chi_phi_goc = sqft * don_gia_goc
rate_discount = 0.05 if sqft >= 1000 else 0.0
tien_giam = chi_phi_goc * rate_discount
phi_don_rac = (chi_phi_goc * 0.10) if is_nha_cu else 0.0
tong_cong = chi_phi_goc - tien_giam + phi_don_rac

st.divider()

# Hiển thị 3 chỉ số chính bằng st.metric
m1, m2, m3 = st.columns(3)
m1.metric("Diện tích", f"{sqft:,.0f} sqft")
m2.metric("Chiết khấu", f"-${tien_giam:,.2f}", delta=f"-{rate_discount*100:.0f}%" if rate_discount > 0 else "0%")
m3.metric("TỔNG DỰ TOÁN", f"${tong_cong:,.2f}", delta="+10% Dọn rác" if is_nha_cu else "Chuẩn")

st.caption("Báo giá dự toán có giá trị trong 30 ngày kể từ ngày khảo sát.")
# Output giao diện hiển thị: [ Thẻ 1: 1,200 sqft ] | [ Thẻ 2: -$300.00 (-5%) ] | [ Thẻ 3: $6,300.00 (+10% Dọn rác) ]

import pandas as pd

st.title("📋 Sổ Theo Dõi Tiến Độ Công Trình")

try:
    df = pd.read_csv("khach_hang.csv", dtype={"Zip Code": str})
    
    st.write(f"Hiện có **{len(df)}** hồ sơ dự án trong hệ thống.")

# BUỔI 21: TRỰC QUAN HÓA TIẾN ĐỘ BẰNG BIỂU ĐỒ
import streamlit as st
import pandas as pd

st.title("📊 Báo Cáo Tiến Độ Công Trình Tuần Này")

df = pd.read_csv("khach_hang.csv")

# Thống kê số lượng theo trạng thái
thong_ke_trang_thai = df["Trạng thái liên hệ"].value_counts()

c1, c2 = st.columns([1.2, 0.8])

with c1:
    st.subheader("Biểu đồ phân bổ tiến độ")
    st.bar_chart(thong_ke_trang_thai, color="#2D625D")

with c2:
    st.subheader("Số liệu chi tiết")
    st.dataframe(thong_ke_trang_thai.rename("Số lượng"), use_container_width=True)
# Biểu đồ hiển thị sắc nét: Cột "Chưa gọi": 14 | Cột "Hẹn khảo sát": 8 | Cột "Đang thi công": 18

# BUỔI 22: BỘ LỌC ĐỘNG BẰNG THANH BÊN SIDEBAR
import streamlit as st
import pandas as pd

st.title("🔎 Tra Cứu Dự Án Theo Bộ Lọc")

df = pd.read_csv("khach_hang.csv", dtype={"Zip Code": str})

# 1. Bộ lọc trên Sidebar
st.sidebar.header("⚙️ BỘ LỌC TÌM KIẾM")

danh_sach_tt = ["Tất cả"] + list(df["Trạng thái liên hệ"].dropna().unique())
chon_tt = st.sidebar.selectbox("Trạng thái công trình:", danh_sach_tt)

danh_sach_zip = ["Tất cả"] + list(df["Zip Code"].dropna().unique())
chon_zip = st.sidebar.selectbox("Khu vực (Zip Code):", danh_sach_zip)

# 2. Xử lý lọc dữ liệu
df_hien_thi = df.copy()
if chon_tt != "Tất cả":
    df_hien_thi = df_hien_thi[df_hien_thi["Trạng thái liên hệ"] == chon_tt]
if chon_zip != "Tất cả":
    df_hien_thi = df_hien_thi[df_hien_thi["Zip Code"] == chon_zip]

# 3. Hiển thị kết quả lọc
st.write(f"Kết quả: Tìm thấy **{len(df_hien_thi)}** công trình phù hợp.")
st.dataframe(df_hien_thi, use_container_width=True, hide_index=True)

Python 3 (Streamlit + SQLite) · Source Code B23app_b23_form_sqlite.py
# BUỔI 23: FORM NHẬP KHÁCH LƯU TRỰC TIẾP VÀO SQLITE
import streamlit as st
import sqlite3
import pandas as pd

st.title("📝 Thêm Hồ Sơ Khách Hàng Mới")

# 1. Form nhập liệu an toàn
with st.form("form_them_khach"):
    ho_ten = st.text_input("Họ và tên chủ nhà:")
    sdt = st.text_input("Số điện thoại liên hệ:")
    zip_code = st.text_input("Mã bưu điện (Zip Code):", value="77001")
    nut_luu = st.form_submit_button("💾 LƯU VÀO CƠ SỞ DỮ LIỆU")

if nut_luu:
    if ho_ten.strip() != "":
        with sqlite3.connect("cong_trinh.db") as conn:
            cursor = conn.cursor()
            cursor.execute(
                "INSERT INTO KhachHang (ho_ten, sdt, zip_code) VALUES (?, ?, ?)",
                (ho_ten.strip(), sdt.strip(), zip_code.strip())
            )
            conn.commit()
        st.success(f"🎉 Đã lưu thành công khách hàng: **{ho_ten}**!")
    else:
        st.error("Vui lòng nhập họ và tên khách hàng!")

# 2. Đọc lại danh sách khách hàng từ SQLite
st.divider()
st.subheader("Danh sách khách hàng hiện có trong Database")
with sqlite3.connect("cong_trinh.db") as conn:
    df_db = pd.read_sql_query("SELECT id, ho_ten, sdt, zip_code, ngay_tao FROM KhachHang ORDER BY id DESC", conn)
st.dataframe(df_db, use_container_width=True, hide_index=True)
# Output kiểm tra: Điền thông tin -> Bấm Lưu -> Bản ghi lập tức xuất hiện ở dòng đầu tiên của bảng bên dưới!

Python 3 (Streamlit + SQLite) · Source Code Mini-CRM B24mini_crm_app.py
# BUỔI 24: ĐỒ ÁN GIAI ĐOẠN 3 - MINI-CRM HOÀN CHỈNH CHO CÔNG TRƯỜNG
import streamlit as st
import sqlite3
import pandas as pd

st.set_page_config(page_title="Mini-CRM Công Trường", page_icon="🏗️", layout="wide")

st.title("🏗️ MINI-CRM GIÁM SÁT CÔNG TRƯỜNG")

# Điều hướng bằng 3 Tab chức năng
tab_bao_gia, tab_tien_do, tab_them_khach = st.tabs([
    "💰 Báo Giá Lát Sàn", 
    "📊 Theo Dõi Tiến Độ", 
    "➕ Thêm Khách Hàng"
])

# --- TAB 1: BÁO GIÁ NHANH ---
with tab_bao_gia:
    st.subheader("Công Cụ Báo Giá Nhanh")
    c1, c2 = st.columns(2)
    with c1:
        dai = st.number_input("Chiều dài phòng (ft):", value=20.0, min_value=0.0)
        rong = st.number_input("Chiều rộng phòng (ft):", value=15.0, min_value=0.0)
    with c2:
        gia = st.number_input("Đơn giá thi công (USD/sqft):", value=5.0, min_value=0.0)
        nha_cu = st.checkbox("Nhà cũ (+10% phí dọn xà bần)")
    
    sqft = dai * rong
    giam = 0.05 if sqft >= 1000 else 0.0
    tien_goc = sqft * gia
    tong = (tien_goc * (1 - giam)) + ((tien_goc * 0.1) if nha_cu else 0.0)
    
    st.metric("TỔNG DỰ TOÁN BÁO GIÁ", f"${tong:,.2f} USD", delta=f"{sqft:,.0f} sqft")

# --- TAB 2: TIẾN ĐỘ DỰ ÁN ---
with tab_tien_do:
    st.subheader("Tiến Độ Thi Công Các Công Trình")
    with sqlite3.connect("cong_trinh.db") as conn:
        df_khach = pd.read_sql_query("SELECT * FROM KhachHang", conn)
    
    col_chart, col_data = st.columns([1, 2])
    with col_chart:
        st.write("**Phân bố theo Zip Code:**")
        if not df_khach.empty and "zip_code" in df_khach.columns:
            st.bar_chart(df_khach["zip_code"].value_counts())
    with col_data:
        st.write("**Danh sách hồ sơ:**")
        st.dataframe(df_khach, use_container_width=True, hide_index=True)

# --- TAB 3: THÊM KHÁCH HÀNG MỚI ---
with tab_them_khach:
    st.subheader("Nhập Hồ Sơ Khách Hàng Mới")
    with st.form("form_crm"):
        ten = st.text_input("Tên chủ nhà:")
        sdt = st.text_input("Số điện thoại:")
        zip_cd = st.text_input("Zip Code:", value="77001")
        if st.form_submit_button("LƯU HỒ SƠ"):
            if ten.strip():
                with sqlite3.connect("cong_trinh.db") as conn:
                    conn.execute("INSERT INTO KhachHang (ho_ten, sdt, zip_code) VALUES (?, ?, ?)", (ten, sdt, zip_cd))
                    conn.commit()
                st.success(f"✅ Đã thêm hồ sơ khách hàng: {ten}")
            else:
                st.error("Vui lòng điền tên khách hàng!")
# Ứng dụng Mini-CRM sẵn sàng đứng lớp: Chạy lệnh: streamlit run mini_crm_app.py Đầy đủ chức năng Báo giá + Thống kê + Lưu trữ SQLite!
