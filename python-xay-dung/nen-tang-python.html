# BUỔI 3: PHÂN LOẠI QUY MÔ & CHIẾT KHẤU DỰ ÁN
NGUONG_DIEN_TICH = 1000.0   # Ngưỡng sqft hưởng ưu đãi
DON_GIA_GOC = 5.0           # USD / sqft

sqft = float(input("Nhập tổng diện tích công trình (sqft): "))

if sqft >= NGUONG_DIEN_TICH:
    phan_tram_giam = 0.05   # Giảm 5%
    don_gia_thuc = DON_GIA_GOC * (1 - phan_tram_giam)
    tien_tiet_kiem = (sqft * DON_GIA_GOC) * phan_tram_giam
    tong_tien = sqft * don_gia_thuc
    print("✅ ĐỦ ĐIỀU KIỆN ƯU ĐÃI DỰ ÁN LỚN (GIẢM 5%)")
    print(f"Đơn giá sau giảm: {don_gia_thuc:.2f} USD/sqft")
    print(f"Khách tiết kiệm:  {tien_tiet_kiem:,.2f} USD")
else:
    tong_tien = sqft * DON_GIA_GOC
    print("ℹ️ DỰ ÁN TIÊU CHUẨN (Áp dụng đơn giá niêm yết)")
    print(f"Đơn giá áp dụng: {DON_GIA_GOC:.2f} USD/sqft")

print(f">>> TỔNG BÁO GIÁ THI CÔNG: {tong_tien:,.2f} USD")
# Output kiểm thử với sqft = 1200: ✅ ĐỦ ĐIỀU KIỆN ƯU ĐÃI DỰ ÁN LỚN (GIẢM 5%) Đơn giá sau giảm: 4.75 USD/sqft Khách tiết kiệm: 300.00 USD >>> TỔNG BÁO GIÁ THI CÔNG: 5,700.00 USD

# BUỔI 4: BÁO GIÁ ĐA ĐIỀU KIỆN (DISCOUNT + REMOVAL FEE)
dien_tich = float(input("Nhập diện tích công trình (sqft): "))
tra_loi_nha_cu = input("Công trình là nhà cũ cần phá dỡ? (c/k): ").lower()
is_nha_cu = (tra_loi_nha_cu == "c")

don_gia_co_ban = 5.0
chi_phi_goc = dien_tich * don_gia_co_ban

# Quy tắc 1: Giảm giá 5% cho công trình từ 1,000 sqft
giam_gia = 0.0
if dien_tich >= 1000:
    giam_gia = chi_phi_goc * 0.05

# Quy tắc 2: Phụ thu 10% dọn xà bần nếu là nhà cũ (độc lập)
phi_don_rac = 0.0
if is_nha_cu:
    phi_don_rac = chi_phi_goc * 0.10

# Tổng kết
tong_thanh_toan = chi_phi_goc - giam_gia + phi_don_rac

print("\n--- BẢNG KÊ CHI PHÍ CHI TIẾT ---")
print(f"Chi phí gốc ({dien_tich} sqft x $5.0): {chi_phi_goc:,.2f} USD")
if giam_gia > 0:
    print(f"[-] Chiết khấu khối lượng (5%):       -{giam_gia:,.2f} USD")
if phi_don_rac > 0:
    print(f"[+] Phí dọn xà bần nhà cũ (10%):     +{phi_don_rac:,.2f} USD")
print("---------------------------------")
print(f"TỔNG GIÁ TRỊ HỢP ĐỒNG:               {tong_thanh_toan:,.2f} USD")
# Output kiểm thử (sqft = 1200, là nhà cũ): --- BẢNG KÊ CHI PHÍ CHI TIẾT --- Chi phí gốc (1200.0 sqft x $5.0): 6,000.00 USD [-] Chiết khấu khối lượng (5%): -300.00 USD [+] Phí dọn xà bần nhà cũ (10%): +600.00 USD --------------------------------- TỔNG GIÁ TRỊ HỢP ĐỒNG: 6,300.00 USD

# BUỔI 7: ĐỊNH NGHĨA HÀM TÍNH BÁO GIÁ DÙNG LẠI
def tinh_bao_gia_san(dai: float, rong: float, don_gia: float, giam_gia: float = 0.0) -> float:
    """Hàm tính báo giá lát sàn một phòng có chiết khấu tùy chọn."""
    sqft = dai * rong
    tien_goc = sqft * don_gia
    tien_thuc_te = tien_goc * (1.0 - giam_gia)
    return tien_thuc_te

def tao_bao_gia_phong(ten: str, dai: float, rong: float, don_gia: float):
    """Tính toán chi tiết cho 1 phòng kèm chiết khấu nếu > 300 sqft."""
