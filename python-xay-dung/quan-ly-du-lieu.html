    # BUỔI 13: KHỞI TẠO KHO LƯU TRỮ SQLITE CHO CÔNG TRÌNH
import sqlite3

# 1. Kết nối hoặc tự động tạo mới file database cong_trinh.db
conn = sqlite3.connect("cong_trinh.db")
cursor = conn.cursor()

# 2. Tạo bảng KhachHang nếu chưa tồn tại
sql_create_table = """
CREATE TABLE IF NOT EXISTS KhachHang (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    ho_ten TEXT NOT NULL,
    sdt TEXT,
    zip_code TEXT,
    ngay_tao TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
"""
cursor.execute(sql_create_table)
conn.commit()

print("✅ ĐÃ KHỞI TẠO THÀNH CÔNG BẢNG 'KhachHang' TRONG cong_trinh.db!")
conn.close()
    # Output kiểm tra: ✅ ĐÃ KHỞI TẠO THÀNH CÔNG BẢNG 'KhachHang' TRONG cong_trinh.db! File cong_trinh.db đã sẵn sàng trong thư mục dự án.

def them_khach_hang(ho_ten, sdt, zip_code):
    """Hàm thêm khách hàng mới và trả về ID vừa sinh."""

    # BUỔI 16: HỆ THỐNG CẬP NHẬT TIẾN ĐỘ THI CÔNG HOÀN CHỈNH
import sqlite3

def cap_nhat_tien_do(du_an_id, trang_thai_moi):
    """Cập nhật tiến độ công trình: Khảo sát -> Thi công -> Bàn giao."""
    trang_thai_hop_le = ["Khảo sát", "Thi công", "Bàn giao", "Đã thanh lý"]
    if trang_thai_moi not in trang_thai_hop_le:
        print(f"❌ Trạng thái '{trang_thai_moi}' không hợp lệ!")
        return False
        
    with sqlite3.connect("cong_trinh.db") as conn:
        cursor = conn.cursor()
        sql = "UPDATE DuAn SET trang_thai = ? WHERE id = ?"
        cursor.execute(sql, (trang_thai_moi, du_an_id))
        conn.commit()
        if cursor.rowcount > 0:
            print(f"✅ ĐÃ CẬP NHẬT DỰ ÁN #{du_an_id} SANG TRẠNG THÁI: '{trang_thai_moi}'")
            return True
        else:
            print(f"⚠️ Không tìm thấy dự án mang mã ID #{du_an_id}!")
            return False

def xoa_du_an(du_an_id):
    """Xóa an toàn một bản ghi dự án với điều kiện ID."""
    with sqlite3.connect("cong_trinh.db") as conn:
        cursor = conn.cursor()
        cursor.execute("DELETE FROM DuAn WHERE id = ?", (du_an_id,))
        conn.commit()
        print(f"🗑️ Đã xóa dự án #{du_an_id} khỏi hệ thống.")
